# dsh-zalo Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A Cordis bundle `dsh-zalo` that lets people chat with a DeepSeek Harness agent through a Zalo Bot (private chats and groups, text, inbound photos, per-chat and global memory, typing indicator).

**Architecture:** A standalone package `packages/dsh-zalo` shaped like `packages/dsh-telegram`. Channel-agnostic modules are copied from dsh-telegram with string chat ids; Zalo-specific modules (`zalo-api`, `poller`, `inbound`, `render`, `turn`, `bot`) are new. Updates arrive by long polling `getUpdates`; replies go out as markdown chunks of at most 2000 characters.

**Tech Stack:** TypeScript 6 (nodenext, strict), Node >= 22.19 global `fetch`, Cordis 4, `@deepseek-ai/schemastery`, vitest 4, pnpm 10.14.0.

**Spec:** `docs/superpowers/specs/2026-10-15-dsh-zalo-design.md`

## Global Constraints

- Branch: `feat/dsh-zalo`. `packages/dsh-telegram` is not modified.
- Zalo base URL: `https://bot-api.zaloplatforms.com/bot<TOKEN>/<method>`, every call `POST` with a JSON body.
- Every chat id, user id, message id, `allowFrom` and `superAdmins` entry is a `string`.
- Max characters per outbound message: 2000 (`messageSize` default 2000, range 500..2000).
- Group trigger: text contains `@<account_name>` (case-insensitive) or starts with `groupPrefix` (default `/ask`) followed by whitespace or end.
- Dedupe: in-memory set of the last 1000 `message_id`s.
- Typing indicator every `typingIntervalMs` (default 4000); failures only warn.
- No runtime dependency beyond `@deepseek-ai/cordis-plugin-logger-console`, `@deepseek-ai/dsh-brand`, `@deepseek-ai/schemastery`; HTTP uses global `fetch`.
- Log lines are prefixed `dsh-zalo: `. The bot token never appears in a log line or error message.
- Code, comments, file names and commit messages in English; no emoji in code.
- Plugin names: `dsh-zalo` (service `zalo`), `dsh-zalo/logger`. Events: `zalo/turn`, `zalo/message`.
- Env vars: `ZALO_BOT_TOKEN`, `ZALO_ALLOW_FROM`, `ZALO_SUPER_ADMINS`, `DSH_ZALO_WORKSPACE`, `DSH_ZALO_DATA`, `DSH_MODEL`, `DSH_LOG_LEVEL`.

## Review Focus

1. Zalo rejects a markdown chunk (LLM output with tables or unbalanced `*`): the chunk is resent as plain text and the user still gets the answer. Test in Task 9.
2. A long reply with no whitespace, or an emoji sitting on the 2000-char boundary: every chunk is <= 2000 UTF-16 units and no surrogate pair is split. Test in Task 7.
3. A chat or message id containing `/`, `..` or `\`: files stay inside their directory (`pathSafe`). Test in Task 2.
4. `getUpdates` returns the same message twice, or an update without `message`: the message is handled once, the malformed update is skipped. Test in Task 6.
5. A photo URL that fails to download: the turn still runs with an `[image could not be downloaded]` marker. Test in Task 8.

```ts
/** Image types `dsh-attachment` accepts. */
export type ImageMediaType = 'image/png' | 'image/jpeg' | 'image/webp' | 'image/gif'

/** An id made safe as one path segment: anything but letters, digits, `_` and `-` becomes `_`. */
export function pathSafe(id: string): string {
  return id === '' ? '_' : id.replace(/[^A-Za-z0-9_-]/g, '_')
}

/** Media type from the file signature; undefined when the bytes are not a supported image. */
export function sniffImageType(data: Buffer): ImageMediaType | undefined {
  if (data.length >= 8 && data.subarray(0, 8).equals(Buffer.from([0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a]))) return 'image/png'
  if (data.length >= 3 && data[0] === 0xff && data[1] === 0xd8 && data[2] === 0xff) return 'image/jpeg'
  if (data.length >= 12 && data.toString('latin1', 0, 4) === 'RIFF' && data.toString('latin1', 8, 12) === 'WEBP') return 'image/webp'
  if (data.length >= 4 && data.toString('latin1', 0, 4) === 'GIF8') return 'image/gif'
  return undefined
}

const EXTENSIONS: Record<ImageMediaType, string> = { 'image/png': '.png', 'image/jpeg': '.jpg', 'image/webp': '.webp', 'image/gif': '.gif' }

export function extensionFor(type: ImageMediaType): string {
  return EXTENSIONS[type]
}
```

```ts
import { appendFile, mkdir, readFile } from 'node:fs/promises'
import { join } from 'node:path'
import { SerialQueue, writeFileAtomic } from './json-file.ts'
import { pathSafe } from './media.ts'

/** One observed Zalo message; bot replies carry `bot: true`, messages that got a turn carry `handled: true`. `ts` is ISO, so it sorts as text. */
export interface ChatLogEntry {
  ts: string
  message_id: string
  user_id: string
  name: string
  text: string
  media?: string[]
  bot?: true
  handled?: true
}

/** Folds line breaks so text rendered as one line of a block cannot forge another line (e.g. a fake sender). */
export function oneLine(text: string): string {
  return text.replace(/\r?\n/g, ' ')
}

export function senderLabel(entry: Pick<ChatLogEntry, 'user_id' | 'name' | 'bot'>): string {
  if (entry.bot) return 'assistant'
  const base = `id:${oneLine(entry.user_id)}`
  return entry.name === '' ? base : `${base} (${oneLine(entry.name)})`
}

export function formatEntry(entry: ChatLogEntry): string {
  const media = entry.media === undefined ? '' : ` [media: ${entry.media.join(', ')}]`
  return `[${entry.ts}] ${senderLabel(entry)}: ${oneLine(entry.text)}${media}`
}

/** A log is trimmed back to `maxEntries` only once it grows this much past it, so trims stay rare. */
const TRIM_SLACK = 1.2

/**
 * Append-only per-chat JSONL log under `dir`; `onAppend` hears every line once it is written.
 * Readers parse the whole file, so it is trimmed to the newest `maxEntries` to keep each read bounded.
 */
export class ChatLog {
  /** Lines per chat file, counted from disk on the first append after start. */
  private readonly counts = new Map<string, number>()
  private readonly writes = new SerialQueue()

  constructor(
    private readonly dir: string,
    private readonly onAppend?: (chatId: string, entry: ChatLogEntry) => void,
    private readonly maxEntries = Number.POSITIVE_INFINITY,
  ) {}

  private file(chatId: string): string {
    return join(this.dir, `${pathSafe(chatId)}.jsonl`)
  }

  async append(chatId: string, entry: ChatLogEntry): Promise<void> {
    await this.writes.run(async () => {
      await mkdir(this.dir, { recursive: true })
      await appendFile(this.file(chatId), `${JSON.stringify(entry)}\n`, 'utf8')
      const count = this.counts.get(chatId) ?? (await this.readAll(chatId)).length - 1
      this.counts.set(chatId, count + 1)
      if (count + 1 > this.maxEntries * TRIM_SLACK) await this.trim(chatId)
    })
    this.onAppend?.(chatId, entry)
  }

  private async trim(chatId: string): Promise<void> {
    const kept = (await this.readAll(chatId)).slice(-this.maxEntries)
    await writeFileAtomic(this.file(chatId), kept.map(e => `${JSON.stringify(e)}\n`).join(''))
    this.counts.set(chatId, kept.length)
  }

  async readAll(chatId: string): Promise<ChatLogEntry[]> {
    let raw: string
    try {
      raw = await readFile(this.file(chatId), 'utf8')
    } catch (error) {
      if ((error as NodeJS.ErrnoException).code === 'ENOENT') return []
      throw error
    }
    return raw.split('\n').filter(line => line !== '').map(line => JSON.parse(line) as ChatLogEntry)
  }

  /** Entries logged after `afterTs`, minus the trigger, keeping the newest `limit`. Zalo ids carry no order, so time does. */
  async recent(chatId: string, afterTs: string, excludeMessageId: string, limit: number): Promise<ChatLogEntry[]> {
    if (limit === 0) return []
    const entries = (await this.readAll(chatId)).filter(e => e.ts > afterTs && e.message_id !== excludeMessageId)
    return entries.slice(-limit)
  }

  /** Newest `limit` entries (oldest first) logged before `before`, optionally containing `query`. */
  async history(chatId: string, options: { limit: number; before?: string; query?: string }): Promise<ChatLogEntry[]> {
    const needle = options.query?.toLowerCase()
    const entries = (await this.readAll(chatId)).filter(e =>
      (options.before === undefined || e.ts < options.before)
      && (needle === undefined || e.text.toLowerCase().includes(needle)))
    return entries.slice(-options.limit)
  }
}
```

```ts
  private readonly writes = new SerialQueue()

  constructor(private readonly file: string) {}

  async load(): Promise<void> {
    let raw: string
    try {
      raw = await readFile(this.file, 'utf8')
    } catch (error) {
      if ((error as NodeJS.ErrnoException).code === 'ENOENT') return
      throw error
    }
    this.records = new Map(Object.entries(JSON.parse(raw) as Record<string, ChatSessionRecord>))
  }

  get(chatId: string): ChatSessionRecord | undefined {
    return this.records.get(chatId)
  }

  async set(chatId: string, record: ChatSessionRecord): Promise<void> {
    this.records.set(chatId, record)
    await this.flush()
  }

  async delete(chatId: string): Promise<void> {
    this.records.delete(chatId)
    await this.flush()
  }

  /** Chats flush concurrently; each queued write snapshots the map when it runs, so the last one holds every change. */
  private flush(): Promise<void> {
    return this.writes.run(() => writeJsonAtomic(this.file, Object.fromEntries(this.records)))
  }
}
```

```ts
import type { RetryConfig } from './config.ts'

export const ZALO_API_BASE = 'https://bot-api.zaloplatforms.com'

export interface ZaloUser { id: string; display_name?: string; is_bot?: boolean }

/** A message as Zalo delivers it; only the fields of its `event_name` are present. */
export interface ZaloMessage {
  from: ZaloUser
  chat: { id: string; chat_type: 'PRIVATE' | 'GROUP' }
  message_id: string
  /** Milliseconds since the epoch. */
  date: number
  text?: string
  photo?: string
  caption?: string
  sticker?: string
  url?: string
  voice_url?: string
}

export interface ZaloUpdate { event_name: string; message: ZaloMessage }

/** The Bot API surface the plugin needs; tests substitute a recorder. */
export interface ZaloApi {
  getMe(): Promise<{ id: string; accountName: string }>
  /** Long-polls for updates; resolves `[]` when the poll times out empty. */
  getUpdates(timeoutSec: number, signal?: AbortSignal): Promise<ZaloUpdate[]>
  sendMessage(chatId: string, text: string, options?: { parseMode?: 'markdown' }): Promise<{ messageId: string }>
  sendChatAction(chatId: string, action: 'typing'): Promise<void>
  downloadUrl(url: string): Promise<Buffer>
}

export class ZaloApiError extends Error {
  constructor(message: string, readonly code?: number) {
    super(message)
    this.name = 'ZaloApiError'
  }
}

const REQUEST_TIMEOUT_MS = 30_000
const DOWNLOAD_TIMEOUT_MS = 60_000
/** Zalo answers an empty long poll with this code. */
const POLL_TIMEOUT_CODE = 408

const sleep = (ms: number) => new Promise(resolve => setTimeout(resolve, ms))

/** Fetches with exponential backoff on network failures, 429 and 5xx; the last attempt's outcome is returned as is. */
async function fetchWithRetry(retry: RetryConfig, attempt: () => Promise<Response>, signal?: AbortSignal): Promise<Response> {
  for (let n = 1; ; n++) {
    try {
      const response = await attempt()
      const transient = response.status === 429 || response.status >= 500
      if (!transient || n >= retry.maxAttempts) return response
      // Release the discarded response's connection before retrying.
      await response.body?.cancel()
    } catch (error) {
      if (n >= retry.maxAttempts || signal?.aborted === true) throw error
    }
    await sleep(Math.min(retry.startDelayMs * 2 ** (n - 1), retry.maxDelayMs))
  }
}

interface Envelope<T> { ok?: boolean; result?: T; error_code?: number; description?: string }

export function createZaloApi(token: string, retry: RetryConfig, fetchImpl: typeof fetch = fetch): ZaloApi {
  const call = async <T>(method: string, body: object, timeoutMs = REQUEST_TIMEOUT_MS, signal?: AbortSignal): Promise<T | undefined> => {
    const response = await fetchWithRetry(retry, () => fetchImpl(`${ZALO_API_BASE}/bot${token}/${method}`, {
      method: 'POST',
      headers: { 'content-type': 'application/json' },
      body: JSON.stringify(body),
      signal: signal === undefined ? AbortSignal.timeout(timeoutMs) : AbortSignal.any([signal, AbortSignal.timeout(timeoutMs)]),
    }), signal)
    const payload = await response.json().catch(() => undefined) as Envelope<T> | undefined
    if (payload?.ok !== true) {
      throw new ZaloApiError(`${method} failed: ${payload?.description ?? `HTTP ${response.status}`}`, payload?.error_code ?? response.status)
    }
    return payload.result
  }

  return {
    async getMe() {
      const me = await call<{ id: string; account_name: string }>('getMe', {})
      if (me === undefined) throw new ZaloApiError('getMe failed: no result')
      return { id: me.id, accountName: me.account_name }
    },
    async getUpdates(timeoutSec, signal) {
      try {
        const result = await call<ZaloUpdate | ZaloUpdate[] | null>('getUpdates', { timeout: timeoutSec }, (timeoutSec + 10) * 1000, signal)
        if (result === undefined || result === null) return []
        return Array.isArray(result) ? result : [result]
      } catch (error) {
        if (error instanceof ZaloApiError && error.code === POLL_TIMEOUT_CODE) return []
        throw error
      }
    },
    async sendMessage(chatId, text, options = {}) {
      const sent = await call<{ message_id: string }>('sendMessage', {
        chat_id: chatId,
        text,
        ...(options.parseMode === undefined ? {} : { parse_mode: options.parseMode }),
      })
      return { messageId: sent?.message_id ?? '' }
    },
    async sendChatAction(chatId, action) {
      await call('sendChatAction', { chat_id: chatId, action })
    },
    async downloadUrl(url) {
      const response = await fetchWithRetry(retry, () => fetchImpl(url, { signal: AbortSignal.timeout(DOWNLOAD_TIMEOUT_MS) }))
      if (!response.ok) throw new ZaloApiError(`download failed: HTTP ${response.status}`, response.status)
      return Buffer.from(await response.arrayBuffer())
    },
  }
}
```

```ts
import type { ZaloApi, ZaloUpdate } from './zalo-api.ts'

/** Insertion-ordered set of recent message ids; getUpdates has no offset, so redelivery is filtered here. */
export class SeenIds {
  private readonly ids = new Set<string>()

  constructor(private readonly limit: number) {}

  /** True when `id` was not seen yet; it is remembered from now on. */
  add(id: string): boolean {
    if (this.ids.has(id)) return false
    this.ids.add(id)
    if (this.ids.size > this.limit) this.ids.delete(this.ids.values().next().value as string)
    return true
  }
}

export interface PollerOptions {
  api: Pick<ZaloApi, 'getUpdates'>
  timeoutSec: number
  onUpdate: (update: ZaloUpdate) => void
  backoff: { startDelayMs: number; maxDelayMs: number }
  log: { warn(msg: string): void; error(msg: string): void }
  seenLimit?: number
}

export interface Poller { stop(): Promise<void> }

const message = (error: unknown) => (error instanceof Error ? error.message : String(error))

/** Long-polls until stopped; a failure backs off and retries, so the loop never ends on its own. */
export function startPolling(options: PollerOptions): Poller {
  const seen = new SeenIds(options.seenLimit ?? 1000)
  const abort = new AbortController()
  const sleep = (ms: number) => new Promise<void>((resolve) => {
    const timer = setTimeout(resolve, ms)
    abort.signal.addEventListener('abort', () => { clearTimeout(timer); resolve() }, { once: true })
  })

  const loop = (async () => {
    let failures = 0
    while (!abort.signal.aborted) {
      let updates: ZaloUpdate[]
      try {
        updates = await options.api.getUpdates(options.timeoutSec, abort.signal)
        failures = 0
      } catch (error) {
        if (abort.signal.aborted) return
        const delayMs = Math.min(options.backoff.startDelayMs * 2 ** failures, options.backoff.maxDelayMs)
        failures++
        options.log.error(`dsh-zalo: getUpdates failed (${message(error)}); retrying in ${delayMs}ms`)
        await sleep(delayMs)
        continue
      }
      for (const update of updates) {
        const id = update.message?.message_id
        if (typeof id !== 'string') {
          options.log.warn(`dsh-zalo: skipped an update without a message (${update.event_name})`)
          continue
        }
        if (!seen.add(id)) continue
        try {
          options.onUpdate(update)
        } catch (error) {
          options.log.error(`dsh-zalo: update handler failed: ${message(error)}`)
        }
      }
    }
  })()

  return {
    async stop() {
      abort.abort()
      await loop
    },
  }
}
```

```ts
/** Zalo rejects a message longer than this many characters. */
export const ZALO_MAX_MESSAGE_CHARS = 2000

const isHighSurrogate = (code: number) => code >= 0xd800 && code <= 0xdbff

/** Where to cut `window`: a paragraph break, else a line break, else a space, if one sits in its second half; else the hard limit. */
function cutPoint(window: string, max: number): number {
  for (const separator of ['\n\n', '\n', ' ']) {
    const at = window.lastIndexOf(separator)
    if (at >= max / 2) return at
  }
  return isHighSurrogate(window.charCodeAt(max - 1)) ? max - 1 : max
}

/** Splits a reply into messages of at most `maxChars` UTF-16 units, preferring natural breaks. */
export function splitMessage(text: string, maxChars: number): string[] {
  const parts: string[] = []
  let rest = text.trim()
  while (rest.length > maxChars) {
    const cut = cutPoint(rest.slice(0, maxChars), maxChars)
    const part = rest.slice(0, cut).trimEnd()
    if (part !== '') parts.push(part)
    rest = rest.slice(cut).trimStart()
  }
  if (rest !== '') parts.push(rest)
  return parts
}
```

```ts
import { join } from 'node:path'
import type { ChatLogEntry } from './chatlog.ts'
import { extensionFor, pathSafe, sniffImageType, type ImageMediaType } from './media.ts'
import type { ZaloApi, ZaloMessage, ZaloUpdate, ZaloUser } from './zalo-api.ts'

export interface InboundImage { data: Buffer; mediaType: ImageMediaType }

export interface InboundMessage {
  chatId: string
  messageId: string
  isGroup: boolean
  sender: ZaloUser
  text: string
  images: InboundImage[]
  logEntry: ChatLogEntry
}

export interface TriggerOptions { accountName: string; groupPrefix: string }

const DOWNLOAD_FAILED = '[image could not be downloaded]'

/** Text shown for events that carry no text of their own. */
const MARKERS: Record<string, string> = {
  'message.image.received': '[photo]',
  'message.sticker.received': '[sticker]',
  'message.voice.received': '[voice]',
  'message.unsupported.received': '[unsupported message]',
}

const escapeRegExp = (text: string) => text.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
const mentionPattern = (accountName: string) => new RegExp(`@${escapeRegExp(accountName)}`, 'gi')
const prefixPattern = (prefix: string) => new RegExp(`^\\s*${escapeRegExp(prefix)}(?=\\s|$)`, 'i')

export function bodyText(message: ZaloMessage): string {
  return [message.text, message.caption].filter((t): t is string => t !== undefined && t !== '').join('\n')
}

/** A group message addresses the bot when it mentions `@accountName` or starts with the prefix word. */
export function hasTrigger(text: string, options: TriggerOptions): boolean {
  return mentionPattern(options.accountName).test(text) || prefixPattern(options.groupPrefix).test(text)
}

export function stripTrigger(text: string, options: TriggerOptions): string {
  return text
    .replace(prefixPattern(options.groupPrefix), '')
    .replace(mentionPattern(options.accountName), '')
    .replace(/[ \t]+/g, ' ')
    .trim()
}

function isoOf(date: number): string {
  const parsed = new Date(date)
  return Number.isNaN(parsed.getTime()) ? new Date().toISOString() : parsed.toISOString()
}

/** The chat-log line for a message; `media` lists saved file paths. */
export function logEntryFor(update: ZaloUpdate, media?: string[]): ChatLogEntry {
  const { message } = update
  const text = bodyText(message)
  return {
    ts: isoOf(message.date),
    message_id: message.message_id,
    user_id: message.from.id,
    name: message.from.display_name ?? '',
    text: text === '' ? MARKERS[update.event_name] ?? '' : text,
    ...(media === undefined || media.length === 0 ? {} : { media }),
  }
}

/** Downloads the photo and saves it as `inbox/<message_id><ext>`; undefined when it fails or is not an image. */
async function fetchPhoto(url: string, messageId: string, options: { api: ZaloApi; inboxDir: string; log: { warn(msg: string): void } }): Promise<{ image: InboundImage; path: string } | undefined> {
  try {
    const data = await options.api.downloadUrl(url)
    const mediaType = sniffImageType(data)
    if (mediaType === undefined) throw new Error('not a supported image')
    await mkdir(options.inboxDir, { recursive: true })
    const path = join(options.inboxDir, `${pathSafe(messageId)}${extensionFor(mediaType)}`)
    await writeFile(path, data)
    return { image: { data, mediaType }, path }
  } catch (error) {
    options.log.warn(`dsh-zalo: photo of message ${messageId} could not be downloaded (${error instanceof Error ? error.message : String(error)})`)
    return undefined
  }
}

export async function parseInbound(
  update: ZaloUpdate,
  options: TriggerOptions & { api: ZaloApi; inboxDir: string; log: { warn(msg: string): void } },
): Promise<InboundMessage> {
  const { message } = update
  const isGroup = message.chat.chat_type === 'GROUP'
  const photo = message.photo === undefined ? undefined : await fetchPhoto(message.photo, message.message_id, options)
  const logEntry = logEntryFor(update, photo === undefined ? undefined : [photo.path])
  let text = isGroup ? stripTrigger(bodyText(message), options) : bodyText(message)
  if (message.photo !== undefined && photo === undefined) text = text === '' ? DOWNLOAD_FAILED : `${text}\n${DOWNLOAD_FAILED}`
  if (text === '') text = MARKERS[update.event_name] ?? ''
  return {
    chatId: message.chat.id,
    messageId: message.message_id,
    isGroup,
    sender: message.from,
    text,
    images: photo === undefined ? [] : [photo.image],
    logEntry,
  }
}
```

```ts
  outcome: TurnOutcome
  text: string
  /** Id of the last message sent for this turn. */
  sentMessageId?: string
  /** Wall-clock breakdown: agent work versus Zalo delivery. */
  timing: { agentMs: number; deliverMs: number }
}

export interface TurnSummary { text: string; error?: string; aborted: boolean }

/** Last assistant text, any error reason, and whether the turn was cancelled, since `firstSeq`. */
export function summarizeTurn(session: TurnAgent['session'], firstSeq: number): TurnSummary {
  let started = false
  let text = ''
  let error: string | undefined
  let aborted = false
  for (let seq = firstSeq; seq < session.seq; seq++) {
    const event = session.eventAt(seq as SessionSeq)
    if (event === undefined) continue
    if (event.type === 'turn/start') {
      started = true
      continue
    }
    if (!started) continue
    if (event.type === 'assistant/message') {
      const joined = event.data.message.content.flatMap(b => (b.type === 'text' ? [b.text] : [])).join('')
      if (joined !== '') text = joined
    }
    if (event.type === 'turn/end') {
      const reason = event.data.reason as { kind: string; error?: { code: string; message: string } }
      if (reason.kind === 'error' && reason.error !== undefined) error = `${reason.error.code}: ${reason.error.message}`
      if (reason.kind === 'aborted') aborted = true
    }
  }
  return { text, aborted, ...(error === undefined ? {} : { error }) }
}

/** Zalo has no editable status message; the typing indicator repeats until the returned stop is called. Failures warn once. */
export function startTyping(api: ZaloApi, chatId: string, intervalMs: number, log: { warn(msg: string): void }): () => void {
  let warned = false
  const send = () => {
    api.sendChatAction(chatId, 'typing').catch((error: unknown) => {
      if (warned) return
      warned = true
      log.warn(`dsh-zalo: typing indicator failed in chat ${chatId} (${String(error)})`)
    })
  }
  send()
  const timer = setInterval(send, intervalMs)
  return () => clearInterval(timer)
}

/** Sends one chunk as markdown, then as plain text if Zalo rejects the markup; undefined when both fail. */
async function sendPart(api: ZaloApi, chatId: string, part: string, log: TurnOptions['log']): Promise<string | undefined> {
  try {
    return (await api.sendMessage(chatId, part, { parseMode: 'markdown' })).messageId
  } catch (error) {
    log.warn(`dsh-zalo: markdown send rejected in chat ${chatId} (${String(error)}); retrying as plain text`)
  }
  try {
    return (await api.sendMessage(chatId, part)).messageId
  } catch (error) {
    log.error(`dsh-zalo: plain send failed in chat ${chatId} (${String(error)})`)
    return undefined
  }
}

async function notice(options: TurnOptions, text: string): Promise<string | undefined> {
  try {
    return (await options.api.sendMessage(options.chatId, text)).messageId
  } catch (error) {
    options.log.error(`dsh-zalo: notice failed in chat ${options.chatId} (${String(error)})`)
    return undefined
  }
}

type Delivery = Omit<TurnResult, 'timing'>

const withId = (id: string | undefined) => (id === undefined ? {} : { sentMessageId: id })

async function deliver(options: TurnOptions, summary: TurnSummary, timedOut: boolean): Promise<Delivery> {
  if (timedOut) {
    const id = await notice(options, `The reply timed out after ${Math.round(options.turnTimeoutMs / 1000)}s and was stopped.`)
    return { outcome: 'timeout', text: summary.text, ...withId(id) }
  }
  if (summary.error !== undefined) return { outcome: 'error', text: summary.text, ...withId(await notice(options, `Error: ${summary.error}`)) }
  if (summary.aborted) return { outcome: 'stopped', text: summary.text, ...withId(await notice(options, 'Stopped.')) }
  if (summary.text.trim() === '') return { outcome: 'empty', text: '', ...withId(await notice(options, '(no reply)')) }

  let lastId: string | undefined
  for (const part of splitMessage(summary.text, options.messageSize)) {
    const id = await sendPart(options.api, options.chatId, part, options.log)
    if (id === undefined) return { outcome: 'undelivered', text: summary.text, ...withId(lastId) }
    lastId = id
  }
  return { outcome: 'sent', text: summary.text, ...withId(lastId) }
}

export async function runTurn(options: TurnOptions): Promise<TurnResult> {
  const { agent } = options
  const now = options.now ?? Date.now
  const stopTyping = startTyping(options.api, options.chatId, options.typingIntervalMs, options.log)
  const firstSeq = agent.session.seq
  let timedOut = false
  let timer: NodeJS.Timeout | undefined
  const agentStart = now()
  try {
    agent.followup(createUserMessage({ content: options.content, source: { kind: 'user' } }))
    const timeout = new Promise<void>((resolve) => {
      timer = setTimeout(() => {
        timedOut = true
        agent.cancel({ kind: 'user' })
        resolve()
      }, options.turnTimeoutMs)
    })
    await Promise.race([agent.whenIdle(), timeout])
  } finally {
    if (timer !== undefined) clearTimeout(timer)
    stopTyping()
  }
  const agentMs = now() - agentStart

  const summary = summarizeTurn(agent.session, firstSeq)
  const deliverStart = now()
  const delivered = await deliver(options, summary, timedOut)
  return { ...delivered, timing: { agentMs, deliverMs: now() - deliverStart } }
}
```

```ts
export type GateVerdict = 'ignore' | 'log-only' | 'handle'

export interface ImageStore {
  saveImages(images: ReadonlyArray<{ data: Uint8Array; mediaType: ImageMediaType }>): Promise<ReadonlyArray<{ attachmentId: unknown }>>
}

export interface BotDeps {
  api: ZaloApi
  config: Config
  chatLog: ChatLog
  memory: MemoryStore
  agents: ChatAgents
  attachments: ImageStore
  bot: { id: string; accountName: string }
  log: { info(msg: string): void; warn(msg: string): void; error(msg: string): void }
  /** Called when a message is about to get a turn (not for commands or log-only messages). */
  noteTurn?: (chatId: string) => void

/** Ids only: Zalo display names are chosen by the sender and prove nothing. */
export function isAllowed(user: ZaloUser, allowFrom: readonly string[]): boolean {
  return allowFrom.some(entry => entry.trim() === user.id)
}

export function gate(update: ZaloUpdate, options: { allowFrom: readonly string[]; botId: string; accountName: string; groupPrefix: string }): GateVerdict {
  const { message } = update
  const sender = message.from
  if (sender === undefined || sender.is_bot === true || sender.id === options.botId) return 'ignore'
  const allowed = isAllowed(sender, options.allowFrom)
  if (message.chat.chat_type !== 'GROUP') return allowed ? 'handle' : 'ignore'
  return allowed && hasTrigger(bodyText(message), options) ? 'handle' : 'log-only'
}

export type Command = 'reset' | 'stop' | 'help'

const HELP_TEXT = '/reset - start a new conversation\n/stop - cancel the running reply\n/help - show this list'

export function commandOf(text: string): Command | undefined {
  const match = /^\s*\/(reset|stop|help)\s*$/.exec(text)
  return match === null ? undefined : match[1] as Command
}

const gateOptions = (deps: BotDeps) => ({
  allowFrom: deps.config.allowFrom, botId: deps.bot.id, accountName: deps.bot.accountName, groupPrefix: deps.config.groupPrefix,
})

/** Commands are for super admins only; anyone else sending `/stop` gets an ordinary turn. */
function commandFrom(update: ZaloUpdate, deps: BotDeps): Command | undefined {
  if (!deps.config.superAdmins.includes(update.message.from?.id)) return undefined
  return commandOf(stripTrigger(bodyText(update.message), gateOptions(deps)))
}

/** Untrusted group text (other members included) fenced off from the user's own message; the persona names the tag as data. */
function recentBlock(entries: ChatLogEntry[]): string {
  return `<group_messages>\n${entries.map(e => `- ${senderLabel(e)}: ${oneLine(e.text)}`).join('\n')}\n</group_messages>`
}

/** Chat plus global memory for the first turn of a session; undefined when reading failed, so the session stays unmarked. */
async function memoryBlock(deps: BotDeps, chatId: string): Promise<string | undefined> {
  try {
    const [chat, global] = await Promise.all([deps.memory.list({ kind: 'chat', chatId }), deps.memory.list({ kind: 'global' })])
    return renderMemoryBlock(chat, global)
  } catch (error) {
    deps.log.warn(`dsh-zalo: reading memory for chat ${chatId} failed: ${error instanceof Error ? error.message : String(error)}`)
    return undefined
  }
}

const seconds = (ms: number) => `${(ms / 1000).toFixed(1)}s`

async function runCommand(command: Command, chatId: string, deps: BotDeps): Promise<void> {
  if (command === 'reset') {
    await deps.agents.reset(chatId)
    await deps.api.sendMessage(chatId, 'Started a new conversation.')
  } else if (command === 'stop') {
    // A cancelled turn reports "Stopped." on its own; only an idle agent needs a reply here.
    if (!deps.agents.stop(chatId)) await deps.api.sendMessage(chatId, 'Nothing is running.')
  } else {
    await deps.api.sendMessage(chatId, HELP_TEXT)
  }
}

export async function handleMessage(update: ZaloUpdate, deps: BotDeps): Promise<void> {
  const verdict = gate(update, gateOptions(deps))
  if (verdict === 'ignore') return
  const chatId = update.message.chat.id
  if (verdict === 'log-only') {
    await deps.chatLog.append(chatId, logEntryFor(update))
    return
  }
  const command = commandFrom(update, deps)
  if (command !== undefined) {
    await runCommand(command, chatId, deps)
    return
  }
  deps.noteTurn?.(chatId)

  const receivedAt = Date.now()
  const lagMs = Math.max(0, receivedAt - update.message.date)
  const inbound = await parseInbound(update, {
    api: deps.api, inboxDir: join(deps.agents.workspaceFor(chatId), 'inbox'), log: deps.log,
    accountName: deps.bot.accountName, groupPrefix: deps.config.groupPrefix,
  })
  await deps.chatLog.append(chatId, { ...inbound.logEntry, handled: true })

  const resolved = await deps.agents.resolve(chatId)
  if (resolved.resumeFailed !== undefined) {
    await deps.api.sendMessage(chatId, 'The previous conversation could not be restored; starting a new one.')
  }
  deps.agents.setTurn(chatId, { sender: inbound.sender, isGroup: inbound.isGroup })

  let text = inbound.text
  if (inbound.isGroup) {
    const recent = await deps.chatLog.recent(chatId, deps.agents.lastTurnAt(chatId), inbound.messageId, deps.config.recentMessagesLimit)
    if (recent.length > 0) text = `${recentBlock(recent)}\n\n${text}`
  }
  // The memory block goes in front of the first user message of a session; later turns rely on memory_recall.
  const sessionId = String(resolved.agent.id)
  const memoryText = deps.agents.memoryInjected(chatId, sessionId) ? '' : await memoryBlock(deps, chatId)
  if (memoryText !== undefined && memoryText !== '') text = `${memoryText}\n\n${text}`
  await deps.agents.markTurn(chatId, inbound.logEntry.ts)

  const content: ContentBlock[] = [{ type: 'text', text }]
  if (inbound.images.length > 0) {
    const refs = await deps.attachments.saveImages(inbound.images)
    for (const ref of refs) content.push({ type: 'image', attachment: ref as never })
  }
  const prepMs = Date.now() - receivedAt
  if (memoryText !== undefined) await deps.agents.markMemoryInjected(chatId, sessionId)

  const result = await runTurn({
    api: deps.api,
    agent: resolved.agent as never,
    chatId,
    content,
    messageSize: deps.config.messageSize,
    typingIntervalMs: deps.config.typingIntervalMs,
    turnTimeoutMs: deps.config.turnTimeoutMs,
    log: deps.log,
  })
  if (result.text !== '') {
    await deps.chatLog.append(chatId, {
      ts: new Date().toISOString(),
      message_id: result.sentMessageId ?? '',
      user_id: deps.bot.id,
      name: deps.bot.accountName,
      text: result.text,
      bot: true,
    })
  }
  deps.log.info(`dsh-zalo: chat ${chatId} message ${inbound.messageId} -> ${result.outcome}`
    + ` (lag ${seconds(lagMs)}, prep ${seconds(prepMs)}, agent ${seconds(result.timing.agentMs)}, deliver ${seconds(result.timing.deliverMs)})`)
}

export interface Dispatcher {
  dispatch(update: ZaloUpdate): void
  /** Appends a task to the chat's queue; tasks of one chat never overlap. */
  enqueue(chatId: string, task: () => Promise<void>): void
  /** Whether the chat has a running or waiting task. */
  isBusy(chatId: string): boolean
}

/**
 * Serialises message handling per chat so turns never overlap; commands bypass
 * the queue so `/stop` and `/reset` act on the turn that is running. A failing
 * task is logged and never blocks the next one.
 */
export function createDispatcher(deps: BotDeps): Dispatcher {
  const chains = new Map<string, Promise<void>>()
  const report = (update: ZaloUpdate) => (error: unknown) => {
    deps.log.error(`dsh-zalo: message ${update.message.message_id} in chat ${update.message.chat.id} failed: ${error instanceof Error ? error.stack ?? error.message : String(error)}`)
  }
  const enqueue = (chatId: string, task: () => Promise<void>): void => {
    const previous = chains.get(chatId) ?? Promise.resolve()
    const next = previous
      .then(task)
      .catch((error: unknown) => {
        deps.log.error(`dsh-zalo: queued task in chat ${chatId} failed: ${error instanceof Error ? error.stack ?? error.message : String(error)}`)
      })
      .finally(() => {
        if (chains.get(chatId) === next) chains.delete(chatId)
      })
    chains.set(chatId, next)
  }
  return {
    dispatch(update) {
      if (gate(update, gateOptions(deps)) === 'handle' && commandFrom(update, deps) !== undefined) {
        void handleMessage(update, deps).catch(report(update))
        return
      }
      enqueue(update.message.chat.id, () => handleMessage(update, deps).catch(report(update)))
    },
    enqueue,
    isBusy: chatId => chains.has(chatId),
  }
}
```

```ts
export function apply(ctx: Context, config: Config): void {
  assertConfig(config)
  const log = loggerFor(ctx)
  ctx.effect(() => {
    const timer = startRetention(config, log)
    return () => clearInterval(timer)
  }, 'dsh-zalo: file retention')

  let revoke: (() => void) | undefined

    const cancel = retryStart(() => start(ctx, config), (result) => {
      if (stopped) {
        void result.poller.stop().then(() => result.agents.disposeAll())
        return
      }
      started = result
      // Provided only now: Cordis starts dependants as soon as a name is provided, with or without a value.
      revoke = ctx.provide('zalo', result.service)
    }, (error, delayMs) => {
      log.error(`dsh-zalo: failed to start: ${error instanceof Error ? error.stack ?? error.message : String(error)}; retrying in ${delayMs}ms`)
    }, STARTUP_BACKOFF)
    return () => {
      stopped = true
      cancel()
      if (started === undefined) return
      const { poller, agents } = started
      started = undefined
      // Dependants stop first, while the bot can still serve their last calls.
      revoke?.()
      revoke = undefined
      void poller.stop().then(() => agents.disposeAll())
    }
  }, 'dsh-zalo: bot runner')
}
```

```dockerfile
FROM node:22-bookworm-slim AS builder
```
